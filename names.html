Compare Any Name to “Braden”

data = FileAttachment("data/comparison_names.csv").csv({ typed: true })

viewof searchName = Inputs.text({
  label: "Search a name to compare:",
  value: "Brandon",
  placeholder: "Type a name (e.g., Brandon, Brayden, Liam, Emma, Noah)..."
})

cleaned = searchName.trim().toLowerCase()
displayName = searchName.trim() ? searchName.trim().charAt(0).toUpperCase() + searchName.trim().slice(1).toLowerCase() : "Comparison"

filtered = data.filter(d => 
  d.name.toLowerCase() === cleaned || 
  d.name === "Braden"
)

Plot.plot({
  title: `Popularity Comparison: "${displayName}" vs. "Braden" (1924–2023)`,
  y: { grid: true, label: "Total Births" },
  x: { tickFormat: "d", label: "Year" },
  color: { 
    legend: true,
    domain: ["Braden", displayName],
    range: ["#0284c7", "#ec4899"]
  },
  marks: [
    Plot.lineY(filtered, {
      x: "year", 
      y: "count", 
      stroke: d => d.name === "Braden" ? "Braden" : displayName, 
      strokeWidth: 2.5
    }),
    Plot.tip(filtered, Plot.pointerX({
      x: "year", 
      y: "count", 
      title: d => `${d.name} (${d.year}): ${d.count.toLocaleString()} births`
    }))
  ]
})
